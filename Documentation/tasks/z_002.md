```csharp
[HttpGet]
public async Task<ActionResult<IEnumerable<ProductTransactionDto>>> GetProductTransactions()
{
    List<ProductTransactionDto> transactions = await _context.ProductTransactions
        .Include(t => t.Product)
        .Include(t => t.Batch)
        .OrderByDescending(t => t.Timestamp)
        .Select(t => new ProductTransactionDto(
            t.Id,
            t.Product.Id,
            t.Product.Name,
            t.Batch != null ? t.Batch.Id : null,
            t.Quantity,
            t.TransactionType.ToString(),
            t.Timestamp
        ))
        .ToListAsync();

    return Ok(transactions);
}
```
